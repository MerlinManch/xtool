cmake_minimum_required(VERSION 3.21)
project(xtool_windows_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(xtool-linker linker.cpp)
add_executable(xtool-codesign codesign.cpp)
find_package(LibArchive REQUIRED)
find_package(LibLZMA REQUIRED)
add_executable(xtool-unxip unxip.cpp)
add_executable(xtool-zip zip.cpp)
target_link_libraries(xtool-zip PRIVATE LibArchive::LibArchive)
target_link_libraries(xtool-unxip PRIVATE LibArchive::LibArchive LibLZMA::LibLZMA)
add_executable(xtool-test-child ../tests/child.cpp)
if(WIN32)
  add_library(xtool-test-websocket SHARED ../../Sources/CWindowsWebSocket/WindowsWebSocket.c)
  target_include_directories(xtool-test-websocket PRIVATE ../../Sources/CWindowsWebSocket/include)
  target_link_libraries(xtool-test-websocket PRIVATE winhttp)
  set_target_properties(xtool-test-websocket PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/websocket-tests")
  add_executable(xtool-test-support ../tests/windows_support.c ../../Sources/CWindowsSupport/WindowsSupport.c)
  target_include_directories(xtool-test-support PRIVATE ../../Sources/CWindowsSupport/include)
  target_link_libraries(xtool-test-support PRIVATE crypt32)
endif()
if(MSVC)
  target_compile_options(xtool-test-websocket PRIVATE /W4 /WX)
  foreach(tool xtool-linker xtool-unxip xtool-zip xtool-codesign)
    target_sources(${tool} PRIVATE ../xtool.exe.manifest)
  endforeach()
  target_compile_options(xtool-linker PRIVATE /W4 /WX)
  target_compile_options(xtool-unxip PRIVATE /W4 /WX)
  target_compile_options(xtool-zip PRIVATE /W4 /WX)
endif()
install(TARGETS xtool-linker xtool-unxip xtool-codesign xtool-zip RUNTIME DESTINATION bin)
